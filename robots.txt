# robots.txt
# ----------
# Instrucciones basicas para crawlers.
#
# IMPORTANTE:
# robots.txt controla principalmente RASTREO.
# No es un mecanismo de privacidad y no reemplaza noindex.

# / representa todo el sitio: permitimos su rastreo.
Allow: /
